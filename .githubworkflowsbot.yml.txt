name: Crypto Signal Bot

on:
  schedule:
    - cron: '17 * * * *'  # Her saatin 17. dakikası (Kimse bunu kullanmaz, anında çalışır!)
  workflow_dispatch:

permissions:
  contents: write # GitHub deposuna yazma ve güncelleme izni verir

jobs:
  run-bot:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.10'

      - name: Install dependencies
        run: |
          pip install pandas numpy requests openpyxl

      - name: Run bot script
        run: |
          python bot.py

      - name: Commit and push changes if new signals found
        run: |
          git config --global user.name "github-actions[bot]"
          git config --global user.email "github-actions[bot]@users.noreply.github.com"
          git add sent_signals.json
          git diff --quiet && git diff --staged --quiet || (git commit -m "Update sent signals history [skip ci]" && git push)
